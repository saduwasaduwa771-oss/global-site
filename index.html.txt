/* ---------- Scroll reveal ---------- */
const io=new IntersectionObserver(entries=>{
  entries.forEach(e=>{if(e.isIntersecting){e.target.classList.add('visible');io.unobserve(e.target);}});
},{threshold:.12});
document.querySelectorAll('.reveal').forEach(el=>io.observe(el));

  const c=COURSES[key];
  document.getElementById('modalTitle').textContent=c.title;
  document.getElementById('modalHero').style.backgroundImage=`url('${c.img}')`;

  document.getElementById('modalSchedule').textContent=c.schedule;
  document.getElementById('modalCert').textContent=c.cert;
  document.getElementById('modalEntry').textContent=c.entry;
  document.getElementById('modalCareer').textContent=c.career;
  document.getElementById('modalTraining').textContent=c.training;
  document.getElementById('courseModal').classList.add('open');
  document.body.style.overflow='hidden';
}
function closeCourse(){
  document.getElementById('courseModal').classList.remove('open');
  document.body.style.overflow='';
}
document.getElementById('courseModal').addEventListener('click',e=>{
  if(e.target.id==='courseModal')closeCourse();
});
document.addEventListener('keydown',e=>{
  if(e.key==='Escape'){closeCourse();closeLightbox();}
});

/* ---------- Lightbox ---------- */
function openLightbox(item){
  const img=item.querySelector('img');
  document.getElementById('lightboxImg').src=img.src.replace('w=700','w=1400').replace('w=1000','w=1600');
  document.getElementById('lightboxImg').alt=img.alt;
  document.getElementById('lightbox').classList.add('open');
  document.body.style.overflow='hidden';
}
function closeLightbox(){
  document.getElementById('lightbox').classList.remove('open');
  document.body.style.overflow='';
}

/* ---------- Form validation ---------- */
const form=document.getElementById('regForm');
function setErr(id,bad){document.getElementById(id).closest('.field').classList.toggle('error',bad);return !bad;}
form.addEventListener('submit',e=>{
  e.preventDefault();
  let ok=true;
  ok=setErr('fullname',document.getElementById('fullname').value.trim().length<3)&&ok;
  ok=setErr('phone',!/^[0-9+\-\s()]{9,15}$/.test(document.getElementById('phone').value.trim()))&&ok;
  ok=setErr('email',!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(document.getElementById('email').value.trim()))&&ok;
  ok=setErr('course',!document.getElementById('course').value)&&ok;
  ok=setErr('address',document.getElementById('address').value.trim().length<5)&&ok;
  if(!ok){form.querySelector('.field.error input, .field.error select')?.focus();return;}
  document.getElementById('formFields').style.display='none';
  document.getElementById('formSuccess').classList.add('show');
});
['fullname','phone','email','course','address'].forEach(id=>{
  document.getElementById(id).addEventListener('input',()=>document.getElementById(id).closest('.field').classList.remove('error'));
  document.getElementById(id).addEventListener('change',()=>document.getElementById(id).closest('.field').classList.remove('error'));
});
function resetForm(){
  form.reset();
  document.getElementById('formFields').style.display='';
  document.getElementById('formSuccess').classList.remove('show');
}
